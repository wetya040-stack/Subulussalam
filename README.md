    const grid = document.getElementById("answers-grid");
    const template = document.getElementById("answer-card-template");
    const emptyState = document.getElementById("empty-state");
    const searchInput = document.getElementById("search-input");
    const visibleCount = document.getElementById("visible-count");
    const togglePatternsButton = document.getElementById("toggle-patterns");
    let activeFilter = "all";
    let patternsVisible = true;

    function makeCard(item, index) {
      const fragment = template.content.cloneNode(true);
      const card = fragment.querySelector("article");
      card.dataset.questionId = item.no;
      card.dataset.type = item.type;
      card.style.animation = `rise .45s ease both ${Math.min(index * 0.025, 0.35)}s`;
      fragment.querySelector(".question-badge").textContent = String(item.no).padStart(2, "0");
      fragment.querySelector(".question-type").textContent = item.type === "huruf" ? "Deret Huruf" : "Deret Angka";
      fragment.querySelector(".series-value").textContent = item.series;
      fragment.querySelector(".answer-value").textContent = item.answer;
      fragment.querySelector(".choice-value").textContent = item.choice;
      fragment.querySelector(".pattern-value").textContent = item.pattern;
      return fragment;
    }

    function renderAnswers() {
      const keyword = searchInput.value.trim().toLowerCase();
      const filtered = answers.filter((item) => {
        const isTypeMatch = activeFilter === "all" || item.type === activeFilter;
        const searchable = `${item.no} ${item.series} ${item.answer} ${item.choice} ${item.pattern}`.toLowerCase();
        return isTypeMatch && searchable.includes(keyword);
      });

      grid.innerHTML = "";
      const cardFragment = document.createDocumentFragment();
      filtered.forEach((item, index) => cardFragment.appendChild(makeCard(item, index)));
      grid.appendChild(cardFragment);

      visibleCount.textContent = filtered.length;
      emptyState.classList.toggle("hidden", filtered.length !== 0);
      grid.classList.toggle("pattern-hidden", !patternsVisible);

      if (window.lucide) window.lucide.createIcons();
    }

    document.querySelectorAll("[data-filter]").forEach((button) => {
      button.addEventListener("click", () => {
        activeFilter = button.dataset.filter;
        document.querySelectorAll("[data-filter]").forEach((item) => {
          item.classList.toggle("active", item === button);
        });
        renderAnswers();
      });
    });

    searchInput.addEventListener("input", renderAnswers);

    togglePatternsButton.addEventListener("click", () => {
      patternsVisible = !patternsVisible;
      togglePatternsButton.setAttribute("aria-pressed", String(patternsVisible));
      togglePatternsButton.querySelector('[data-template-id="hide-patterns-label"]').classList.toggle("hidden", !patternsVisible);
      togglePatternsButton.querySelector('[data-template-id="show-patterns-label"]').classList.toggle("hidden", patternsVisible);
      renderAnswers();
    });

    document.getElementById("print-button").addEventListener("click", () => window.print());

    renderAnswers();
    if (window.lucide) window.lucide.createIcons();
